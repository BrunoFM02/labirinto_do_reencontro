function generateMaze() { mazeMap = []; for (let row = 0; row < ROWS; row++) { mazeMap[row] = []; for (let col = 0; col < COLS; col++) { mazeMap[row][col] = "#"; } } /* ========================= CRIA O LABIRINTO BASE ========================= */ function carve(row, col) { mazeMap[row][col] = "."; const directions = shuffle([ [-2, 0], [2, 0], [0, -2], [0, 2] ]); for (const direction of directions) { const dr = direction[0]; const dc = direction[1]; const newRow = row + dr; const newCol = col + dc; if ( newRow > 0 && newRow < ROWS - 1 && newCol > 0 && newCol < COLS - 1 && mazeMap[newRow][newCol] === "#" ) { mazeMap[row + dr / 2][col + dc / 2] = "."; carve(newRow, newCol); } } } carve(1, 1); mazeMap[1][1] = "S"; /* ========================= ABRIR CAMINHOS ALTERNATIVOS Remove algumas paredes internas para criar bifurcações e rotas diferentes. ========================= */ const attempts = 35; for (let i = 0; i < attempts; i++) { const row = Math.floor(Math.random() * (ROWS - 2)) + 1; const col = Math.floor(Math.random() * (COLS - 2)) + 1; /* Só abre paredes */ if (mazeMap[row][col] !== "#") { continue; } const up = mazeMap[row - 1][col] !== "#"; const down = mazeMap[row + 1][col] !== "#"; const left = mazeMap[row][col - 1] !== "#"; const right = mazeMap[row][col + 1] !== "#"; /* Cria conexões horizontais */ if (left && right) { mazeMap[row][col] = "."; } /* Cria conexões verticais */ else if (up && down) { mazeMap[row][col] = "."; } /* Algumas bifurcações */ else if ( Math.random() < 0.35 && ( (up && left) || (up && right) || (down && left) || (down && right) ) ) { mazeMap[row][col] = "."; } } /* ========================= ENCONTRAR O PONTO MAIS DISTANTE DO INÍCIO A SAÍDA CONTINUA SENDO O FINAL MAIS LONGE. ========================= */ const visited = []; for (let row = 0; row < ROWS; row++) { visited[row] = []; for (let col = 0; col < COLS; col++) { visited[row][col] = false; } } const queue = []; queue.push({ row: 1, col: 1, distance: 0 }); visited[1][1] = true; let farthest = { row: 1, col: 1, distance: 0 }; while (queue.length > 0) { const current = queue.shift(); if (current.distance > farthest.distance) { farthest = current; } const directions = [ [-1, 0], [1, 0], [0, -1], [0, 1] ]; for (const direction of directions) { const newRow = current.row + direction[0]; const newCol = current.col + direction[1]; if ( newRow >= 0 && newRow < ROWS && newCol >= 0 && newCol < COLS && !visited[newRow][newCol] && mazeMap[newRow][newCol] !== "#" ) { visited[newRow][newCol] = true; queue.push({ row: newRow, col: newCol, distance: current.distance + 1 }); } } } /* ========================= DEFINIR A SAÍDA ========================= */ mazeMap[farthest.row][farthest.col] = "E"; /* ========================= RESETAR JOGADOR ========================= */ playerRow = 1; playerCol = 1; gameFinished = false; moving = false; }
